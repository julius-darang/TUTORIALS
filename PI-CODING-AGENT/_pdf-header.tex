% Small geometric mark for the cover: a transparent, corner-forward cube
% drawn as a white wireframe so it follows the cover's black-and-white palette.
\newcommand{\picubesingle}[2]{%
  \begin{scope}[xshift=#1cm,yshift=#2cm]
    % A small 10-degree yaw around the vertical (Y) axis. The nearest
    % vertical edge stays fixed while the two horizontal directions shift.
    \def\yaw{-10}
    \pgfmathsetmacro{\cubeleftx}{-1.5*cos(\yaw) + 1.5*sin(\yaw)}
    \pgfmathsetmacro{\cubelefty}{0.675*cos(\yaw) + 0.675*sin(\yaw)}
    \pgfmathsetmacro{\cuberightx}{1.5*sin(\yaw) + 1.5*cos(\yaw)}
    \pgfmathsetmacro{\cuberighty}{-0.675*sin(\yaw) + 0.675*cos(\yaw)}
    \pgfmathsetmacro{\cubebackx}{\cubeleftx + \cuberightx}
    \pgfmathsetmacro{\cubebacky}{\cubelefty + \cuberighty}

    % The nearest vertical edge is centered, with a shallow diamond top face.
    \coordinate (back-top)     at (\cubebackx,\cubebacky);
    \coordinate (left-top)     at (\cubeleftx,\cubelefty);
    \coordinate (front-top)    at (0,0);
    \coordinate (right-top)    at (\cuberightx,\cuberighty);
    \coordinate (back-bottom)  at (\cubebackx,\cubebacky-1.65);
    \coordinate (left-bottom)  at (\cubeleftx,\cubelefty-1.65);
    \coordinate (front-bottom) at (0,-1.65);
    \coordinate (right-bottom) at (\cuberightx,\cuberighty-1.65);

    % Faces remain unfilled and transparent; only the visible edges are drawn.
    \draw[white!72] (back-top) -- (left-top) -- (front-top) --
      (right-top) -- cycle;
    \draw[white!72] (left-top) -- (left-bottom) -- (front-bottom);
    \draw[white!72] (right-top) -- (right-bottom) -- (front-bottom);
    \draw[white!72] (left-bottom) -- (front-bottom) -- (right-bottom);
    \draw[white!72] (front-top) -- (front-bottom);
  \end{scope}
}

\newcommand{\picube}{%
  \begin{center}
    \begin{tikzpicture}[xscale=1.15, yscale=1.25, line join=round,
      line cap=round, line width=0.9pt]
      % Nine matching cubes with a small, consistent gap between them.
      \foreach \x in {0,2.55,5.10} {
        \foreach \y in {0,-2.55,-5.10} {
          \picubesingle{\x}{\y}
        }
      }
    \end{tikzpicture}
  \end{center}
}
